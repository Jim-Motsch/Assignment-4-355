def _initial_password(username):
    # Passwords are supplied via environment (VERADEMO_PASSWORD_<USERNAME>)
    # or randomly generated; no credentials are stored in source code.
    env_name = "VERADEMO_PASSWORD_" + re.sub(r"[^A-Za-z0-9]", "_", username).upper()
    return os.environ.get(env_name) or secrets.token_urlsafe(24)
